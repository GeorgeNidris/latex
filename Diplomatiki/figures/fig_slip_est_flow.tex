\begin{tikzpicture}[
    node distance=0.55cm and 1.0cm,
    block/.style   ={rectangle, draw=black, fill=white,      text width=5.2cm,
                     align=center, rounded corners=4pt, minimum height=0.9cm, font=\small},
    io/.style      ={rectangle, draw=black, fill=black!5,    text width=5.2cm,

    outbox/.style  ={rectangle, draw=black, fill=black!12,   text width=5.2cm,
                     align=center, rounded corners=2pt, minimum height=0.9cm, font=\small\bfseries},
    decision/.style={diamond,   draw=black, fill=white,      text width=2.4cm,
                     align=center, font=\scriptsize, inner sep=1pt},
    arr/.style     ={->, thick, black}
]

  \node[io]                       (inp)    {Είσοδοι: γεωμετρία πίστας, ταχύτητα};
  \node[block, below=of inp]      (resamp) {Επαναδειγματοληψία σε ομοιόμορφο πλέγμα};
  \node[block, below=of resamp]   (kinem)  {Κινηματική προεπεξεργασία και εξομάλυνση};
  \node[block, below=of kinem]    (fz)     {Κατανομή κατακόρυφων φορτίων ανά τροχό};
  \node[block, below=of fz]       (solve)  {Επίλυση ισορροπίας για γωνία διεύθυνσης \\ και πλευρική ολίσθηση αμαξώματος};
  \node[decision, below=of solve] (conv)   {σύγκλιση;};
  \node[block, below=of conv]     (wheels) {Υπολογισμός ταχυτήτων ανά τροχό};
  \node[block, below=of wheels]   (slip)   {Υπολογισμός γωνιών και λόγων ολίσθησης};
  \node[outbox, below=of slip]    (out)    {Έξοδος: ολισθήσεις και δυνάμεις ανά τροχό};

  \draw[arr] (inp)    -- (resamp);
  \draw[arr] (resamp) -- (kinem);
  \draw[arr] (kinem)  -- (fz);
  \draw[arr] (fz)     -- (solve);
  \draw[arr] (solve)  -- (conv);
  \draw[arr] (conv)   -- node[right, font=\scriptsize]{ναι} (wheels);
  \draw[arr] (wheels) -- (slip);
  \draw[arr] (slip)   -- (out);

  % failure branch
  \node[font=\scriptsize, right=0.8cm of conv, align=center]
    (fail) {επισήμανση\\ασύγκλιτου σημείου};
  \draw[arr, dashed] (conv.east) -- (fail.west)
      node[midway, above, font=\scriptsize]{όχι};

\end{tikzpicture}
